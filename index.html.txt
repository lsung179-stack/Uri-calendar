  const r=document.documentElement.style

function startWS(){
  const ws=new WebSocket(`${SB}/realtime/v1/websocket?apikey=${SK}&vsn=1.0.0`)
  ws.onopen=()=>{ws.send(JSON.stringify({topic:'realtime:public',event:'phx_join',payload:{config:{postgres_changes:[{event:'*',schema:'public',table:'events',filter:`room_id=eq.${myRoomId}`},{event:'*',schema:'public',table:'members',filter:`room_id=eq.${myRoomId}`}]}},ref:'1'}));setSync('ok')}
  ws.onmessage=async e=>{try{const msg=JSON.parse(e.data);if(msg.payload?.data){setSync('ing');await loadAll();renderTabs();if(view==='monthly')renderMonthly();else renderWeekly();if(selDate)renderDpList();setSync('ok')}}catch(_){}}
  ws.onerror=()=>setSync('err')
  ws.onclose=()=>{setSync('err');setTimeout(startWS,3000)}
  setInterval(()=>{if(ws.readyState===1)ws.send(JSON.stringify({topic:'phoenix',event:'heartbeat',payload:{},ref:'hb'}))},25000)
}

function updateTodayBtn(){

  const appOn=document.getElementById('appWrap').classList.contains('show')
  btn.style.display=(appOn&&view==='monthly'&&!(curY===NOW.getFullYear()&&curM===NOW.getMonth()))?'block':'none'
  // 테마 accentL 색상 가져와서 밝기에 따라 글자색 결정
  const aL=getComputedStyle(document.documentElement).getPropertyValue('--aL').trim()
  btn.style.background=aL
  // hex를 rgb로 변환해서 밝기 계산
  const hex=aL.replace('#','')
  if(hex.length===6){
    const r=parseInt(hex.slice(0,2),16),g=parseInt(hex.slice(2,4),16),b=parseInt(hex.slice(4,6),16)
    const brightness=(r*299+g*587+b*114)/1000
    btn.style.color=brightness>160?'#3a3020':'#ffffff'
  }
}

function buildCells(y,m){
  const fd=new Date(y,m,1).getDay(),dt=new Date(y,m+1,0).getDate()
  const pm=m===0?11:m-1,py=m===0?y-1:y,dp=new Date(py,pm+1,0).getDate()
  const nm=m===11?0:m+1,ny=m===11?y+1:y,c=[]
  for(let i=0;i<fd;i++)c.push({y:py,m:pm,d:dp-fd+1+i,dim:true})
  for(let d=1;d<=dt;d++)c.push({y,m,d,dim:false})
  let n=1;while(c.length%7!==0)c.push({y:ny,m:nm,d:n++,dim:true})
  return c
}
function fillPanel(el,y,m,ue){
  const g=document.createElement('div');g.className='mgrid'
  buildCells(y,m).forEach(c=>{
    const key=dk(c.y,c.m,c.d),dow=new Date(c.y,c.m,c.d).getDay()
    const isT=sameDay(new Date(c.y,c.m,c.d),NOW),arr=ue[key]||[],isH=!!HOLIDAYS[key]&&!c.dim
    const cell=document.createElement('div')
    cell.className=['mc',c.dim?'dim':'',isT?'tod':'',dow===0?'sun':'',dow===6?'sat':'',isH?'hol':''].filter(Boolean).join(' ')
    cell.onclick=()=>{if(!c.dim)openDay(key,c.y,c.m,c.d,dow)}
    const dn=document.createElement('div');dn.className='dn';dn.textContent=c.d;cell.appendChild(dn)
    if(isH){const h=document.createElement('div');h.className='eb ehol';h.textContent=HOLIDAYS[key];cell.appendChild(h)}
    arr.slice(0,2).forEach(ev=>{
      const prevKey=dk(...prevDate(c.y,c.m,c.d))
      const nextKey=dk(...nextDate(c.y,c.m,c.d))
      const hasPrev=(ue[prevKey]||[]).some(e=>e.title===ev.title&&e.color===ev.color)
      const hasNext=(ue[nextKey]||[]).some(e=>e.title===ev.title&&e.color===ev.color)
      const b=document.createElement('div');b.className='eb '+cPill(ev.color)
      if(hasPrev&&hasNext){
        b.textContent='‧';b.style.cssText='border-radius:0;margin-left:-4px;margin-right:-4px;overflow:visible'
      }else if(!hasPrev&&hasNext){
        b.textContent=ev.title;b.style.cssText='border-radius:3px 0 0 3px;margin-right:-4px;overflow:visible'
      }else if(hasPrev&&!hasNext){
        b.textContent='';b.style.cssText='border-radius:0 3px 3px 0;margin-left:-4px;overflow:visible;min-width:6px'
      }else{
        b.textContent=ev.title
      }
      cell.appendChild(b)
    })
    if(arr.length>2){const x=document.createElement('div');x.className='em';x.textContent=`+${arr.length-2}`;cell.appendChild(x)}
    g.appendChild(cell)
  })
  el.innerHTML='';el.appendChild(g)
}
function prevDate(y,m,d){const dt=new Date(y,m,d-1);return[dt.getFullYear(),dt.getMonth(),dt.getDate()]}
function nextDate(y,m,d){const dt=new Date(y,m,d+1);return[dt.getFullYear(),dt.getMonth(),dt.getDate()]}
function resetSlider(){
  const sl=document.getElementById('monthlySlider')
  const prev=adjM(curY,curM,-1),next=adjM(curY,curM,1)
  fillPanel(document.getElementById('panelPrev'),prev.y,prev.m,evs[curUser]||{})
  fillPanel(document.getElementById('panelCur'),curY,curM,evs[curUser]||{})
  fillPanel(document.getElementById('panelNext'),next.y,next.m,evs[curUser]||{})
  sl.classList.remove('anim');sl.style.transform='translateX(-33.333%)';sl.offsetHeight
}

function renderMiniCal(){
  const wrap=document.getElementById('miniCal')

  const cal=document.createElement('div');cal.className='mini-cal'

      if(sameDay(new Date(c.y,c.m,c.d),NOW))el.classList.add('todm')
    }

    if(dateOpt==='multi'){

      if(rangeS&&rangeE){

      }else if(rangeS&&key===rangeS){

function getDatesFromRange(s,e){
  const dates=[]
  const cur=new Date(s+'T00:00:00')
  const end=new Date(e+'T00:00:00')
  while(cur<=end){
    dates.push(dk(cur.getFullYear(),cur.getMonth(),cur.getDate()))
    cur.setDate(cur.getDate()+1)
  }
  return dates
}
