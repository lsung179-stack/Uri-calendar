    const existingMembers=await sb(`members?room_id=eq.${code}`)
    const existing=existingMembers.find(m=>m.name===name)

    }

    // 새 멤버로 참여
    const memberId=genId()
    const color=MEMBER_COLORS[existingMembers.length%MEMBER_COLORS.length]
    await sb('members','POST',{id:memberId,room_id:code,name,color,is_owner:false})

function buildCells(y,m){
  const fd=new Date(y,m,1).getDay(),dt=new Date(y,m+1,0).getDate()
  const pm=m===0?11:m-1,py=m===0?y-1:y,dp=new Date(py,pm+1,0).getDate()
  const nm=m===11?0:m+1,ny=m===11?y+1:y
  const c=[]
  for(let i=0;i<fd;i++)c.push({y:py,m:pm,d:dp-fd+1+i,dim:true})
  for(let d=1;d<=dt;d++)c.push({y,m,d,dim:false})
  let n=1;while(c.length%7!==0)c.push({y:ny,m:nm,d:n++,dim:true})
  return c
}

;(()=>{
  const targets=[document.getElementById('monthlyWrap'),document.getElementById('weeklyWrap')]
  let sx=0,sy=0,st=0,swiping=false,dragging=false
  function onStart(e){if(isSliding)return;const t=e.touches?e.touches[0]:e;sx=t.clientX;sy=t.clientY;st=Date.now();swiping=true;dragging=false}
  function onMove(e){
    if(!swiping||view!=='monthly'||isSliding)return
    const t=e.touches?e.touches[0]:e,dx=t.clientX-sx,dy=t.clientY-sy
    if(!dragging&&Math.abs(dx)>8&&Math.abs(dx)>Math.abs(dy))dragging=true
    if(!dragging)return
    const slider=document.getElementById('monthlySlider')
    slider.style.transition='none'
    slider.style.transform=`translateX(${-33.333+(dx/window.innerWidth*100)}%)`
  }
  function onEnd(e){
    if(!swiping)return;swiping=false
    const t=e.changedTouches?e.changedTouches[0]:e,dx=t.clientX-sx,dy=t.clientY-sy,dt=Date.now()-st
    if(view==='monthly'&&dragging){
      dragging=false
      const slider=document.getElementById('monthlySlider')
      if(Math.abs(dx)>50&&dt<500&&Math.abs(dy)<Math.abs(dx)){
        const d=dx<0?1:-1;curM+=d;if(curM>11){curM=0;curY++}if(curM<0){curM=11;curY--};renderMonthly()
      }else{slider.style.transition='transform 0.28s cubic-bezier(.4,0,.2,1)';slider.style.transform='translateX(-33.333%)'}
      return
    }
    dragging=false
    if(Math.abs(dx)<40||Math.abs(dy)>Math.abs(dx)*0.6||dt>400)return
    if(document.getElementById('dayPanel').classList.contains('on'))return
    if(view==='weekly'){curWeekStart=addDays(curWeekStart,(dx<0?1:-1)*7);renderWeekly()}
  }
  targets.forEach(el=>{if(!el)return;el.addEventListener('touchstart',onStart,{passive:true});el.addEventListener('touchmove',onMove,{passive:true});el.addEventListener('touchend',onEnd,{passive:true})})
})()
