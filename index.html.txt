function buildCells(y,m){
  const fd=new Date(y,m,1).getDay(),dt=new Date(y,m+1,0).getDate()
  const pm=m===0?11:m-1,py=m===0?y-1:y,dp=new Date(py,pm+1,0).getDate()
  const nm=m===11?0:m+1,ny=m===11?y+1:y,c=[]
  for(let i=0;i<fd;i++)c.push({y:py,m:pm,d:dp-fd+1+i,dim:true})
  for(let d=1;d<=dt;d++)c.push({y,m,d,dim:false})
  let n=1;while(c.length%7!==0)c.push({y:ny,m:nm,d:n++,dim:true})
  return c
}

/* 같은 멤버 + 같은 title/color/time + 연속 날짜 → 그룹으로 감지 */
function findEventGroup(memberId,evId,title,color,time){
  const all=[]
  const memberEvs=evs[memberId]||{}
  Object.entries(memberEvs).forEach(([d,list])=>{
    list.forEach(e=>all.push({...e,date:d}))
  })
  const t=time||null
  const matched=all.filter(e=>e.title===title&&e.color===color&&(e.time||null)===t)
  if(matched.length<=1)return [matched.find(e=>e.id===evId)||{id:evId}]
  matched.sort((a,b)=>a.date.localeCompare(b.date))
  const idx=matched.findIndex(e=>e.id===evId)
  if(idx===-1)return [{id:evId}]
  const group=[matched[idx]]
  // 이전 방향
  for(let i=idx-1;i>=0;i--){
    const pd=new Date(group[0].date+'T00:00:00')
    const cd=new Date(matched[i].date+'T00:00:00')
    const diff=Math.round((pd-cd)/86400000)
    if(diff===1)group.unshift(matched[i]);else break
  }
  // 이후 방향
  for(let i=idx+1;i<matched.length;i++){
    const ld=new Date(group[group.length-1].date+'T00:00:00')
    const cd=new Date(matched[i].date+'T00:00:00')
    const diff=Math.round((cd-ld)/86400000)
    if(diff===1)group.push(matched[i]);else break
  }
  return group
}

async function doEditSave(){
  const title=document.getElementById('editTitle').value.trim()
  if(!title){toast('일정 이름을 입력해주세요');return}
  const time=document.getElementById('editTime').value
  setSync('ing')
  try{
    const isCurrentlyShared=!!editSharedId
    const wantShared=editTargetMode==='all'&&members.length>1&&editSelectedMembers.length>1
    
    // ═══════════════════════════════════════════════
    // [Case 1] 함께 → 함께 (멤버 변경 가능)
    // ═══════════════════════════════════════════════
    if(isCurrentlyShared&&wantShared){
      // 같은 shared_id를 가진 현재 멤버들 찾기
      const oldMemberIds=[]
      for(const mid in evs){
        if(evs[mid][editEvDate]?.some(e=>e.shared_id===editSharedId)){
          oldMemberIds.push(mid)
        }
      }
      const oldSet=new Set(oldMemberIds)
      const newSet=new Set(editSelectedMembers)
      const toKeep=oldMemberIds.filter(m=>newSet.has(m))      // 유지 - PATCH
      const toRemove=oldMemberIds.filter(m=>!newSet.has(m))   // 제거 - DELETE
      const toAdd=editSelectedMembers.filter(m=>!oldSet.has(m)) // 추가 - POST
      
      const promises=[]
      const newRows=[]
      // PATCH (유지 멤버 메타만)
      for(const mid of toKeep){
        const ev=evs[mid][editEvDate].find(e=>e.shared_id===editSharedId)
        if(ev)promises.push(api(`events?id=eq.${ev.id}`,'PATCH',{title,time:time||null,color:editSelColor}))
      }
      // DELETE (제거 멤버)
      for(const mid of toRemove){
        const ev=evs[mid][editEvDate].find(e=>e.shared_id===editSharedId)
        if(ev)promises.push(api(`events?id=eq.${ev.id}`,'DELETE'))
      }
      // POST (추가 멤버)
      for(const mid of toAdd){
        const id=genId()
        newRows.push({memberId:mid,id})
        promises.push(api('events','POST',{id,room_id:myRoomId,member_id:mid,date:editEvDate,title,time:time||null,color:editSelColor,shared_id:editSharedId}))
      }
      await Promise.all(promises)
      // 로컬 업데이트
      for(const mid of toKeep){
        if(evs[mid][editEvDate]){
          evs[mid][editEvDate]=evs[mid][editEvDate].map(e=>
            e.shared_id===editSharedId?{...e,title,time:time||null,color:editSelColor}:e
          )
        }
      }
      for(const mid of toRemove){
        if(evs[mid][editEvDate]){
          evs[mid][editEvDate]=evs[mid][editEvDate].filter(e=>e.shared_id!==editSharedId)
          if(!evs[mid][editEvDate].length)delete evs[mid][editEvDate]
        }
      }
      for(const r of newRows){
        if(!evs[r.memberId])evs[r.memberId]={}
        if(!evs[r.memberId][editEvDate])evs[r.memberId][editEvDate]=[]
        evs[r.memberId][editEvDate].push({id:r.id,title,time:time||null,color:editSelColor,shared_id:editSharedId})
      }
      closeOv('editMo')
      if(view==='monthly')renderMonthly();else renderWeekly()
      renderDpList()
      setSync('ok')
      toast('함께 일정이 수정됐어요 ✓')
      return
    }
    
    // ═══════════════════════════════════════════════
    // [Case 2] 함께 → 혼자
    // ═══════════════════════════════════════════════
    if(isCurrentlyShared&&!wantShared){
      // 같은 shared_id를 가진 모든 row 삭제 후 본인만 새 일정 추가
      await api(`events?shared_id=eq.${editSharedId}&date=eq.${editEvDate}`,'DELETE')
      // 로컬 제거
      for(const mid in evs){
        if(evs[mid][editEvDate]){
          evs[mid][editEvDate]=evs[mid][editEvDate].filter(e=>e.shared_id!==editSharedId)
          if(!evs[mid][editEvDate].length)delete evs[mid][editEvDate]
        }
      }
      // 본인 새 일정 1개 POST (shared_id 없이)
      const newId=genId()
      await api('events','POST',{id:newId,room_id:myRoomId,member_id:myId,date:editEvDate,title,time:time||null,color:editSelColor})
      if(!evs[myId])evs[myId]={}
      if(!evs[myId][editEvDate])evs[myId][editEvDate]=[]
      evs[myId][editEvDate].push({id:newId,title,time:time||null,color:editSelColor})
      closeOv('editMo')
      if(view==='monthly')renderMonthly();else renderWeekly()
      renderDpList()
      setSync('ok')
      toast('혼자 일정으로 변경됐어요 ✓')
      return
    }
    
    // ═══════════════════════════════════════════════
    // [Case 3] 혼자 → 함께
    // ═══════════════════════════════════════════════
    if(!isCurrentlyShared&&wantShared){
      const newSharedId='shr_'+Date.now().toString(36)+'_'+Math.random().toString(36).slice(2,8)
      // 처리할 날짜 결정: editScope='all'+그룹이 여러 개면 모든 날짜, 아니면 현재 날짜만
      const datesToProcess=(editScope==='all'&&editGroup.length>1)
        ?editGroup.map(g=>({date:g.date,evId:g.id}))
        :[{date:editEvDate,evId:editEvId}]
      const promises=[]
      const newRowsByMember={}  // {mid:[{id,date},...]}
      for(const {date,evId} of datesToProcess){
        // 본인 row PATCH (제목/시간/색상 + shared_id 추가)
        promises.push(api(`events?id=eq.${evId}`,'PATCH',{title,time:time||null,color:editSelColor,shared_id:newSharedId}))
        // 다른 멤버에게 POST (본인 제외)
        for(const mid of editSelectedMembers){
          if(mid===myId)continue
          const id=genId()
          if(!newRowsByMember[mid])newRowsByMember[mid]=[]
          newRowsByMember[mid].push({id,date})
          promises.push(api('events','POST',{id,room_id:myRoomId,member_id:mid,date,title,time:time||null,color:editSelColor,shared_id:newSharedId}))
        }
      }
      await Promise.all(promises)
      // 로컬 업데이트: 본인 모든 날짜 row PATCH
      for(const {date,evId} of datesToProcess){
        if(evs[myId][date]){
          evs[myId][date]=evs[myId][date].map(e=>
            e.id===evId?{...e,title,time:time||null,color:editSelColor,shared_id:newSharedId}:e
          )
        }
      }
      // 다른 멤버 row 추가 (날짜별)
      for(const mid in newRowsByMember){
        if(!evs[mid])evs[mid]={}
        for(const {id,date} of newRowsByMember[mid]){
          if(!evs[mid][date])evs[mid][date]=[]
          evs[mid][date].push({id,title,time:time||null,color:editSelColor,shared_id:newSharedId})
        }
      }
      closeOv('editMo')
      if(view==='monthly')renderMonthly();else renderWeekly()
      renderDpList()
      setSync('ok')
      const msg=datesToProcess.length>1
        ?`${datesToProcess.length}일 모두 함께 일정으로 변경됐어요 ✓`
        :'함께 일정으로 변경됐어요 ✓'
      toast(msg)
      return
    }
    
    // ═══════════════════════════════════════════════
    // [Case 4] 혼자 → 혼자 (기존 로직 - 그룹 처리 + 단일 처리)
    // ═══════════════════════════════════════════════
    if(editScope==='all'&&editGroup.length>1){
      // 그룹 전체 수정 (기간도 변경 가능)
      const newStart=editNewStart||editGroupStart
      const newEnd=editNewEnd||editGroupEnd
      const oldDates=editGroup.map(g=>g.date)
      const newDates=getDatesFromRange(newStart,newEnd)
      const oldSet=new Set(oldDates)
      const newSet=new Set(newDates)
      const toKeep=editGroup.filter(g=>newSet.has(g.date))
      const toDelete=editGroup.filter(g=>!newSet.has(g.date))
      const toAdd=newDates.filter(d=>!oldSet.has(d))
      const promises=[]
      const newEvents=[]
      for(const g of toKeep)promises.push(api(`events?id=eq.${g.id}`,'PATCH',{title,time:time||null,color:editSelColor}))
      for(const g of toDelete)promises.push(api(`events?id=eq.${g.id}`,'DELETE'))
      for(const d of toAdd){
        const id=genId()
        newEvents.push({id,date:d})
        promises.push(api('events','POST',{id,room_id:myRoomId,member_id:myId,date:d,title,time:time||null,color:editSelColor}))
      }
      await Promise.all(promises)
      for(const g of toKeep){
        if(evs[myId]?.[g.date]){
          evs[myId][g.date]=evs[myId][g.date].map(e=>e.id===g.id?{...e,title,time:time||null,color:editSelColor}:e)
        }
      }
      for(const g of toDelete){
        if(evs[myId]?.[g.date]){
          evs[myId][g.date]=evs[myId][g.date].filter(e=>e.id!==g.id)
          if(!evs[myId][g.date].length)delete evs[myId][g.date]
        }
      }
      for(const ne of newEvents){
        if(!evs[myId])evs[myId]={}
        if(!evs[myId][ne.date])evs[myId][ne.date]=[]
        evs[myId][ne.date].push({id:ne.id,title,time:time||null,color:editSelColor})
      }
      closeOv('editMo')
      if(view==='monthly')renderMonthly();else renderWeekly()
      if(selDate&&(newSet.has(selDate)||oldSet.has(selDate)))renderDpList()
      setSync('ok')
      toast(`${newDates.length}일 일정으로 수정됐어요 ✓`)
    }else{
      // 단일 일정 수정 (날짜 변경 가능)
      const newDate=editNewDate||editEvDate
      await api(`events?id=eq.${editEvId}`,'PATCH',{title,time:time||null,color:editSelColor,date:newDate})
      if(evs[myId]&&evs[myId][editEvDate]){
        evs[myId][editEvDate]=evs[myId][editEvDate].filter(e=>e.id!==editEvId)
        if(!evs[myId][editEvDate].length)delete evs[myId][editEvDate]
      }
      if(!evs[myId])evs[myId]={}
      if(!evs[myId][newDate])evs[myId][newDate]=[]
      evs[myId][newDate].push({id:editEvId,title,time:time||null,color:editSelColor})
      closeOv('editMo')
      if(selDate===editEvDate||selDate===newDate)renderDpList()
      if(view==='monthly')renderMonthly();else renderWeekly()
      setSync('ok');toast('수정됐어요 ✓')
    }
  }catch(e){toast('수정 실패: '+(e.message||'').slice(0,30));setSync('err')}
}

function setTarget(t){
  if(t==='all' && members.length>1){

      if(sameDay(new Date(c.y,c.m,c.d),NOW))el.classList.add('todm')
    }
    el.textContent=c.d

    if(dateOpt==='multi'){

      if(rangeS&&rangeE){

      }
    }

function getDatesFromRange(s,e){
  const dates=[]
  const cur=new Date(s+'T00:00:00')
  const end=new Date(e+'T00:00:00')
  while(cur<=end){
    dates.push(dk(cur.getFullYear(),cur.getMonth(),cur.getDate()))
    cur.setDate(cur.getDate()+1)
  }
  return dates
}

async function saveEv(){
  const title=document.getElementById('newTitle').value.trim()
  if(!title){toast('일정 이름을 입력해주세요');return}
  const time=document.getElementById('newTime').value

  // 저장할 날짜 목록 결정
  let dates=[]
  if(dateOpt==='single') dates=[selDate]
  else if(dateOpt==='range'){
    if(!rangeS){toast('날짜를 선택해주세요');return}
    dates=getDatesFromRange(rangeS,rangeE||rangeS)
  }else if(dateOpt==='multi'){
    if(!multiDates.length){toast('날짜를 선택해주세요');return}
    dates=[...multiDates].sort()
  }

  // 대상 멤버 결정 (혼자 또는 선택된 멤버들)
  const isShared = targetMode==='all' && members.length>1 && selectedMembers.length>1
  const targetMembers = isShared ? [...selectedMembers] : [myId]
  // 함께 일정이면 그룹 식별용 shared_id 생성
  const sharedId = isShared ? ('shr_'+Date.now().toString(36)+'_'+Math.random().toString(36).slice(2,8)) : null

  setSync('ing')
  try{
    // 모든 (멤버 × 날짜) 조합에 대해 병렬 POST
    const promises=[]
    const newEvents=[]  // 로컬 evs 업데이트용
    for(const memberId of targetMembers){
      for(const date of dates){
        const id=genId()
        newEvents.push({memberId,date,id})
        const body={id,room_id:myRoomId,member_id:memberId,date,title,time:time||null,color:selColor}
        if(sharedId)body.shared_id=sharedId
        promises.push(api('events','POST',body))
      }
    }
    await Promise.all(promises)
    // 로컬 evs 업데이트
    for(const ne of newEvents){
      if(!evs[ne.memberId])evs[ne.memberId]={}
      if(!evs[ne.memberId][ne.date])evs[ne.memberId][ne.date]=[]
      const localEv={id:ne.id,title,time:time||null,color:selColor}
      if(sharedId)localEv.shared_id=sharedId
      evs[ne.memberId][ne.date].push(localEv)
    }
    if(view==='monthly')renderMonthly();else renderWeekly()
    renderDpList()
    document.getElementById('newTitle').value='';document.getElementById('newTime').value='12:00'
    showForm=false;document.getElementById('dpForm').classList.remove('on');document.querySelector('.dps').classList.remove('fopen');document.getElementById('dpAddBtn').style.display=''
    const dpbot=document.querySelector('.dpbot');if(dpbot)dpbot.scrollTop=0
    dateOpt='single';rangeS=null;rangeE=null;multiDates=[];targetMode='self';selectedMembers=[myId]
    setSync('ok')
    // 토스트 메시지
    if(isShared){
      toast(dates.length>1?`함께 일정 ${dates.length}일 저장됐어요 ✓`:'함께 일정이 저장됐어요 ✓')
    }else{
      toast(dates.length>1?`${dates.length}일 일정이 저장됐어요 ✓`:'저장됐어요 ✓')
    }
  }catch(e){toast('저장 실패');setSync('err')}
}
async function delEv(evId,idx){
  const ev=evs[myId]?.[selDate]?.[idx]
  if(!ev)return
  // 함께 일정인 경우 확인 후 모두 삭제
  if(ev.shared_id){
    if(!confirm('이 일정은 함께 일정입니다. 모든 멤버에게서 삭제됩니다. 진행할까요?'))return
    setSync('ing')
    try{
      // 같은 shared_id 모든 row 삭제 (단, 같은 날짜의 것만 — 기간 일정은 하루씩 삭제하는 일관성 유지)
      // 안전을 위해 shared_id + date로 필터링
      await api(`events?shared_id=eq.${ev.shared_id}&date=eq.${selDate}`,'DELETE')
      // 로컬 evs: 모든 멤버의 selDate에서 해당 shared_id 제거
      for(const mid in evs){
        if(evs[mid][selDate]){
          evs[mid][selDate]=evs[mid][selDate].filter(e=>e.shared_id!==ev.shared_id)
          if(!evs[mid][selDate].length)delete evs[mid][selDate]
        }
      }
      if(view==='monthly')renderMonthly();else renderWeekly()
      renderDpList();setSync('ok');toast('함께 일정이 삭제됐어요 ✓')
    }catch(e){toast('삭제 실패');setSync('err')}
    return
  }
  // 일반 일정: 본인 것만 삭제
  setSync('ing')
  try{
    await api(`events?id=eq.${evId}`,'DELETE')
    if(evs[myId]&&evs[myId][selDate]){evs[myId][selDate].splice(idx,1);if(!evs[myId][selDate].length)delete evs[myId][selDate]}
    if(view==='monthly')renderMonthly();else renderWeekly()
    renderDpList();setSync('ok')
  }catch(e){toast('삭제 실패');setSync('err')}
}

      const sess=loadSess();if(sess){sess.name=newName;saveSess(sess)}
      nameChanged=true
      setSync('ok')
    }
    // [2] 사진 변경 처리 (로컬 저장)
    if(pendingAvData==='remove'){
      const a=getAvatars();delete a[myId];localStorage.setItem(AKEY,JSON.stringify(a))
      photoChanged=true
    }else if(pendingAvData){
      saveAvLocal(myId,pendingAvData)
      photoChanged=true
    }
    closeOv('avatarMo')
    // 화면 갱신
    if(nameChanged||photoChanged){
      renderTabs()
      // 이름 변경 시 멤버 모달이 다시 열릴 때를 위해 캘린더도 다시 (아바타 이니셜 등에 영향)
      if(view==='monthly')renderMonthly();else renderWeekly()
      if(selDate)renderDpList()
    }
    // 토스트
    if(nameChanged&&photoChanged)toast('프로필이 저장됐어요 ✓')
    else if(nameChanged)toast('이름이 변경됐어요 ✓')
    else if(photoChanged)toast(pendingAvData==='remove'?'사진이 삭제됐어요':'사진이 저장됐어요 ✓')
  }catch(e){
    setSync('err')
    toast('저장 실패: '+(e.message||'').slice(0,30))
  }
}

function openYM(){
  const yList=document.getElementById('ymY'),mList=document.getElementById('ymM')
  yList.innerHTML='';mList.innerHTML=''
  for(let y=NOW.getFullYear()-5;y<=NOW.getFullYear()+5;y++){
    const item=document.createElement('div');item.className='ymi'+(y===curY?' sel':'');item.textContent=y+'년'
    item.onclick=()=>{curY=y;document.querySelectorAll('#ymY .ymi').forEach(x=>x.classList.remove('sel'));item.classList.add('sel');if(view==='monthly')renderMonthly();else renderWeekly();closeOv('ymMo')}
    yList.appendChild(item);if(y===curY)setTimeout(()=>item.scrollIntoView({block:'center'}),50)
  }
  for(let m=0;m<12;m++){
    const item=document.createElement('div');item.className='ymi'+(m===curM?' sel':'');item.textContent=(m+1)+'월'
    item.onclick=()=>{curM=m;document.querySelectorAll('#ymM .ymi').forEach(x=>x.classList.remove('sel'));item.classList.add('sel');if(view==='monthly')renderMonthly();else renderWeekly();closeOv('ymMo')}
    mList.appendChild(item);if(m===curM)setTimeout(()=>item.scrollIntoView({block:'center'}),50)
  }
  document.getElementById('ymMo').classList.add('on')
}

;(()=>{
  const targets=[document.getElementById('monthlyWrap'),document.getElementById('weeklyWrap')]
  let sx=0,sy=0,st=0,sw=false,dr=false
  function onS(e){
    // 시트(dayPanel)가 열려있으면 캘린더 스와이프 비활성화
    if(document.getElementById('dayPanel').classList.contains('on')){sw=false;return}
    const t=e.touches?e.touches[0]:e;sx=t.clientX;sy=t.clientY;st=Date.now();sw=true;dr=false
  }
  function onM(e){
    if(!sw||view!=='monthly')return
    if(document.getElementById('dayPanel').classList.contains('on')){sw=false;return}
    const t=e.touches?e.touches[0]:e,dx=t.clientX-sx,dy=t.clientY-sy
    if(!dr){if(Math.abs(dx)>10&&Math.abs(dx)>Math.abs(dy)*1.5)dr=true;else if(Math.abs(dy)>10){sw=false;return}}
    if(!dr)return;e.preventDefault()
    const sl=document.getElementById('monthlySlider')
    sl.classList.remove('anim');sl.style.transform=`translateX(${-33.333+(dx/window.innerWidth*100)}%)`
  }
  function onE(e){
    if(!sw)return;sw=false;if(!dr){dr=false;return};dr=false
    const t=e.changedTouches?e.changedTouches[0]:e
    const dx=t.clientX-sx,dy=t.clientY-sy,dt=Date.now()-st
    const sl=document.getElementById('monthlySlider')
    if(view==='monthly'){
      if(Math.abs(dx)>40&&dt<600&&Math.abs(dy)<Math.abs(dx)*.8){
        sl.classList.remove('anim');sl.style.transform='translateX(-33.333%)';sl.offsetHeight
        curM+=dx<0?1:-1;if(curM>11){curM=0;curY++}if(curM<0){curM=11;curY--}
        updateHeader();resetSlider()
      }else{sl.classList.add('anim');sl.style.transform='translateX(-33.333%)'}
    }else if(view==='weekly'&&Math.abs(dx)>40&&Math.abs(dy)<Math.abs(dx)*.6&&dt<400){
      curWeekStart=addDays(curWeekStart,(dx<0?1:-1)*7);renderWeekly()
    }
  }
  targets.forEach(el=>{if(!el)return;el.addEventListener('touchstart',onS,{passive:true});el.addEventListener('touchmove',onM,{passive:false});el.addEventListener('touchend',onE,{passive:true})})

  /* 시트 핸들 바: 아래로 스와이프 = 폼 닫기(폼 열려있을 때) 또는 시트 닫기 */
  const dph=document.querySelector('.dph')
  if(dph){
    let phY=0,phX=0,phDrag=false
    dph.addEventListener('touchstart',e=>{
      phY=e.touches[0].clientY;phX=e.touches[0].clientX;phDrag=true
    },{passive:true})
    dph.addEventListener('touchmove',e=>{
      if(!phDrag)return
      const dy=e.touches[0].clientY-phY,dx=e.touches[0].clientX-phX
      // 가로 움직임이 더 크면 취소
      if(Math.abs(dx)>Math.abs(dy)*1.2){phDrag=false;return}
      // 위로 올리는 건 차단 (다운스와이프만 의미있음)
      if(dy<0)return
      // 시트를 살짝 따라 내려가는 시각 피드백
      const dps=document.querySelector('.dps')
      if(dps){dps.style.transition='none';dps.style.transform=`translateY(${Math.min(dy,200)}px)`}
      e.preventDefault&&e.preventDefault()
    },{passive:false})
    dph.addEventListener('touchend',e=>{
      if(!phDrag)return
      phDrag=false
      const dy=(e.changedTouches[0].clientY-phY)
      const dps=document.querySelector('.dps')
      if(dps){dps.style.transition='transform .2s';dps.style.transform=''}
      // 50px 이상 아래로 내렸으면 시트 전체 닫기 (폼 열려있어도 한 번에 꺼짐)
      if(dy>50){closePanel()}
      // transition 정리
      setTimeout(()=>{if(dps)dps.style.transition=''},250)
    },{passive:true})
  }
})()
